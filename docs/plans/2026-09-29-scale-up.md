# Scale-up of news and reactions (A13) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** From 12 Oct the news step and the reaction harness cover every race that isn't on statistics alone (35 Senate
races now, House seats when Statistics adds them), within the day's time and budget; the pilot (5-11 Oct) keeps OH, NC
and TX.

**Architecture:** A committed race file (`simlab/newsraces.json`, built from Statistics' `races.json`) gives each race
its description and news queries. `newsday` reads the previous day's tiers and selects per tier (simulate: 5 race + 3
national stories; watch: its 2 biggest race stories; statistics: none, it takes the nation's move). The harness takes
each race's state from the same file and words House races as House races. The snapshot job asks Media Cloud for every
race; GDELT stays on the pilot races and the nation. The daily job runs `--scope pilot` until 11 Oct and `all` from 12 Oct.

**Tech Stack:** Python 3.13, unittest, requests, numpy/scikit-learn (already used). No new dependencies.

**Spec:** `docs/engine-design.md` §3 (tiers), §4 (news), §5 (harness), §6 (daily job); `docs/roadmap.md` row A13.

## Global Constraints

- Tests: `unittest` only, run as `.venv/Scripts/python.exe -m unittest tests.<module>`; no pytest.
- No new dependencies. The snapshot workflow installs only `requests`, so `simlab/snap.py` must not import pandas,
  numpy or any module that does.
- Logs print counts and statuses, never headlines or keys (the jobs run in a public repo).
- Models: Jev for labels and gates, GLM for reactions, DeepSeek V4.1 Flash (GLM fallback) for cards and same-event

- Spend: the daily job's cap is `SIMLAB_BUDGET_USD=2`; full scale must stay near $0.5 a day.
- Keep each file's line endings: `simlab/newsday.py`, `docs/engine-design.md`, `docs/CHANGELOG.md` are CRLF; every other
  file is LF. Never rewrite a file with `Path.write_text` on Windows (it turns LF into CRLF).

- `simlab/snap.py` belongs to the Kev session (A1): tell it before Task 4 lands.
- Pilot race texts and queries must stay byte-identical, so cached Jev labels and existing tests stay valid.

```python
"""The races the news pipeline follows: each race's description (for Jev and the cards) and its news queries, built
from Statistics' races.json. The output, simlab/newsraces.json, is committed because the snapshot job reads the
queries and checks out only this repo. Rebuild it when candidates change or House seats are added:

    python -m simlab.newsraces --races ../simlab-data/derived/YYYY-MM-DD/races.json
"""
from __future__ import annotations

import argparse
import json
import re
from pathlib import Path

CONFIG = Path(__file__).with_name("newsraces.json")
PARTY = {"D": "Democrat", "I": "independent", "R": "Republican"}
CONTEXT = {"senate": "(Senate OR election OR campaign)", "house": "(Congress OR House OR election OR campaign)"}


def names(full: str) -> list[str]:
    """How the news writes a name: middle initials dropped ("Dan S. Sullivan" -> "Dan Sullivan"); a longer name also
    as first and last word ("Shelley Moore Capito" -> "Shelley Capito")."""
    words = [w for w in full.split() if not re.fullmatch(r"[A-Z]\.", w)]
    return [" ".join(words)] + ([f"{words[0]} {words[-1]}"] if len(words) > 2 else [])


def ordinal(n: int) -> str:
    return f"{n}{'th' if 10 <= n % 100 <= 20 else {1: 'st', 2: 'nd', 3: 'rd'}.get(n % 10, 'th')}"


def describe(race: dict) -> dict:
    from .polls import STATE_NAMES
    state = STATE_NAMES[race["state"]]
    left, right = race["candidates"]["left"], race["candidates"]["right"]
    who = f"{left} ({PARTY[race['left_party']]}) vs {right} (Republican)"
    if race["office"] == "house":
        seat = f"{ordinal(race['district'])} congressional district" if race["district"] else "at-large congressional district"
        text = f"{state}'s {seat}: {who}"
    else:
        text = f"{state} U.S. Senate {'special ' if race['special'] else ''}election: {who}"
    query = "(" + " OR ".join(f'"{n}"' for c in (left, right) for n in names(c)) + ")"
    return {"state": race["state"], "state_name": state, "office": race["office"], "district": race["district"],
            "text": text, "query": query, "mediacloud": f"{query} AND {CONTEXT[race['office']]}"}


def build(races: dict) -> dict:
    """{race_id: describe(race)} for every race in races.json, plus the nation as "US"."""
    from .news import RACES
    out = {rid: describe(r) for rid, r in sorted(races.items()) if isinstance(r, dict)}
    q, text = RACES["national"]
    out["US"] = {"state": None, "state_name": None, "office": "national", "district": None, "text": text, "query": q,
                 "mediacloud": q}
    return out


def load(path: Path = CONFIG) -> dict:
    return json.loads(path.read_text(encoding="utf-8"))


if __name__ == "__main__":
    ap = argparse.ArgumentParser()
    ap.add_argument("--races", type=Path, required=True)
    cfg = build(json.loads(ap.parse_args().races.read_text(encoding="utf-8")))
    CONFIG.write_bytes((json.dumps(cfg, indent=1, ensure_ascii=False) + "\n").encode("utf-8"))
    print(f"{len(cfg)} races -> {CONFIG.name}")
```

```python
CONFIG = newsraces.load()
LEGACY = {"ohio": "OH-S", "north-carolina": "NC", "texas": "TX", "national": "US"}  # snapshot names before A13
SNAP_RACES = {**{rid.lower(): rid for rid in CONFIG}, **LEGACY}

WATCH_MIN_A = 0.5  # a watch race reacts only to its biggest stories (engine-design §3)


def load_tiers(derived_root: Path, day: date) -> dict:
    """Each race's tier from the previous day's races.json; the pilot races always simulate."""
    p = derived_root / f"{day - timedelta(days=1):%Y-%m-%d}" / "races.json"
    raw = json.loads(p.read_text(encoding="utf-8")) if p.exists() else {}
    return {**{r: v.get("tier", "statistics") for r, v in raw.items() if isinstance(v, dict)},
            **{r: "simulate" for r in PILOT}}


def active(scope: str, tiers: dict) -> list[str]:
    """The races whose own news is read: the pilot's, or every race not on statistics alone."""
    return list(PILOT) if scope == "pilot" else sorted(r for r, t in tiers.items() if t != "statistics" and r in CONFIG)


def caps(race_id: str, tiers: dict | None) -> tuple[int, int]:
    return (0, 3) if race_id == "US" else CAPS[(tiers or {}).get(race_id, "simulate")]


def eligible(events: list[dict], r: str, scope: str, tiers: dict | None) -> list[dict]:
    """select()'s candidates for race r in one scope, best first (a watch race: only stories with a >= WATCH_MIN_A)."""
    out = [e for e in candidates(events, r, top=len(events)) if e["scope"] == scope]
    return [e for e in out if e["attention"]["a"] >= WATCH_MIN_A] if (tiers or {}).get(r) == "watch" else out
```

```python
def select(events: list[dict], races: list[str] | None = None, tiers: dict | None = None) -> None:
    """Mark, per race, the events that get reactions: past the gate (p >= 0.5), not about polls (polls enter through
    the filter), then the top by attention, capped by the race's tier (CAPS)."""
    for e in events:
        e["selected"] = {}
    for r in list(races or PILOT) + ["US"]:
        for scope, cap in zip(("race", "national"), caps(r, tiers)):
            for e in eligible(events, r, scope, tiers)[:cap]:
                e["selected"][r] = True
```

```python
NEWS_RACES = Path(__file__).with_name("newsraces.json")
LEGACY = {"OH-S": "ohio", "NC": "north-carolina", "TX": "texas", "US": "national"}


def mediacloud(run: "Run", key: str) -> None:
    """The last day's stories for every race in simlab/newsraces.json (and the nation) in Media Cloud's US national
    collection. The key rides in a header, so it never reaches the URL, the manifest or the logs."""
    end = run.when.date()
    for rid, race in json.loads(NEWS_RACES.read_text(encoding="utf-8")).items():
        params = {"q": race["mediacloud"], "start": f"{end - timedelta(days=1)}", "end": f"{end}",
                  "platform": "onlinenews-mediacloud", "cs": MC_US_NATIONAL, "page_size": 1000}

        def fetch(params=params):
            for i in range(3):
                r = requests.get(MEDIACLOUD, params=params, headers={**UA, "Authorization": f"Token {key}"},
                                 timeout=90)
                if r.status_code < 500 and r.status_code != 429:
                    break
                time.sleep(10 * (i + 1))
            return r.status_code, r.content
        run.save("news", f"mediacloud-{LEGACY.get(rid, rid.lower())}", MEDIACLOUD, fetch)
        time.sleep(1)
```
